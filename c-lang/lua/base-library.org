  // print(...)
  //
  // Receives any number of arguments and prints their values to "stdout",
  // converting each argument to a string following the same rules
  // of "tostring".
  static int luaB_print(lua_State* L) {
      // Total arguments.
      int n = lua_gettop(L);
      int i;
      // For each argument...
      for (i = 1; i <= n; i++) {
          size_t l;
          // Convert argument to string.
          const char* s = luaL_tolstring(L, i, &l);
          // If not the first argument, add a tab before it.
          if (i > 1) {
              lua_writestring("\t", 1);
          }
          // Print the result.
          lua_writestring(s, l);
          // Pop the result.
          lua_pop(L, 1);
      }
      // Add newline.
      lua_writeline();
      return 0;
  }

  // warn(message, ...)
  //
  // Emits a warning message composed of the concatenation of all its arguments.
  static int luaB_warn(lua_State* L) {
      // Total arguments.
      int n = lua_gettop(L);
      int i;
      // Is there at least one argument?
      luaL_checkstring(L, 1);
      // Ensure all elements are strings.
      for (i = 2; i <= n; i++) {
          luaL_checkstring(L, i);
      }
      // Compose warning.
      for (i = 1; i < n; i++) {
          lua_warning(L, lua_tostring(L, i), 1);
      }
      // Close warning.
      lua_warning(L, lua_tostring(L, n), 0);
      return 0;
  }

  #define SPACECHARS " \f\n\r\t\v"

  static const char* b_str2int(const char* s, unsigned base, lua_Integer* pn) {
      lua_Unsigned n = 0;
      int neg = 0;
      // Skip initial whitespace.
      s += strspn(s, SPACECHARS);
      if (*s == '-') {
          s++;
          neg = 1;
      }
      // Handle sign.
      else if (*s == '+') {
          s++;
      }
      // No digit?
      if (!isalnum(cast_uchar(*s))) {
          return NULL;
      }
      do {
          unsigned digit = cast_uint(isdigit(cast_uchar(*s))
                           ? *s - '0'
                           : (toupper(cast_uchar(*s)) - 'A') + 10);
          // Invalid numeral.
          if (digit >= base) {
              return NULL;
          }
          n = n * base + digit;
          s++;
      } while (isalnum(cast_uchar(*s)));
      // Skip trailing whitespace.
      s += strspn(s, SPACECHARS);
      *pn = (lua_Integer)((neg) ? (0u - n) : n);
      return s;
  }

  // tonumber(expression [, radix]) -> number | nil
  //
  // Tries to convert its argument to either an integer or float in base 10.
  // Returns nil if conversion fails.
  //
  // String conversions follow the lexical conventions of Lua.
  //
  // If provided with radix, the provided expression must be a string,
  // which is interpreted as an integer numeral in that base. A radix
  // must be an integer between 2 to 36 inclusive.
  static int luaB_tonumber(lua_State* L) {
      // Standard conversion?
      if (lua_isnoneornil(L, 2)) {
          // Already a number? Return it.
          if (lua_type(L, 1) == LUA_TNUMBER) {
              lua_settop(L, 1);
              return 1;
          } else {
              size_t l;
              const char* s = lua_tolstring(L, 1, &l);
              if (s != NULL && lua_stringtonumber(L, s) == l + 1) {
                  // Successful conversion.
                  return 1;
              }
              // Else not a number (but there must be some parameter)
              luaL_checkany(L, 1);
          }
      } else {
          size_t l;
          const char* s;
          // Avoid warnings.
          lua_Integer n = 0;
          lua_Integer base = luaL_checkinteger(L, 2);
          // No numbers as strings.
          luaL_checktype(L, 1, LUA_TSTRING);
          s = lua_tolstring(L, 1, &l);
          luaL_argcheck(L, 2 <= base && base <= 36, 2, "base out of range");
          if (b_str2int(s, cast_uint(base), &n) == s + l) {
              lua_pushinteger(L, n);
              return 1;
          }
      }
      // Not a number.
      luaL_pushfail(L);
      return 1;
  }

  // error(message [, level])
  //
  // Raises an error with given message as the error object.
  // This function never returns. The level argument specifies
  // where to target error position.
  static int luaB_error(lua_State* L) {
      int level = (int)luaL_optinteger(L, 2, 1);
      lua_settop(L, 1);
      if (lua_type(L, 1) == LUA_TSTRING && level > 0) {
          // Add extra information.
          luaL_where(L, level);
          lua_pushvalue(L, 1);
          lua_concat(L, 2);
      }
      return lua_error(L);
  }

  // getmetatable(object) -> value | metatable | nil
  //
  // Attempts to retrieve a metatable from the given object.
  //
  // __metatable? -> value
  // metatable?   -> metatable
  // otherwise    -> nil
  static int luaB_getmetatable(lua_State* L) {
      luaL_checkany(L, 1);
      if (!lua_getmetatable(L, 1)) {
          lua_pushnil(L);
          return 1;
      }
      luaL_getmetafield(L, 1, "__metatable");
      return 1;
  }

  // setmetatable(table, metatable) -> table
  //
  // Sets the metatable for the given table. If metatable is nil, removes
  // the metatable of the given table. If the original metatable has
  // a __metatable field, raises an error. Returns a table.
  static int luaB_setmetatable(lua_State* L) {
      int t = lua_type(L, 2);
      luaL_checktype(L, 1, LUA_TTABLE);
      luaL_argexpected(L, t == LUA_TNIL || t == LUA_TTABLE, 2, "nil or table");
      if (l_unlikely(luaL_getmetafield(L, 1, "__metatable") != LUA_TNIL)) {
          return luaL_error(L, "cannot change a protected metatable");
      }
      lua_settop(L, 2);
      lua_setmetatable(L, 1);
      return 1;
  }

  // rawequal(x, y) -> boolean
  //
  // Checks whether x is equal to y, without invoking the __eq metamethod for either value.
  // Returns a boolean.
  static int luaB_rawequal(lua_State* L) {
      luaL_checkany(L, 1);
      luaL_checkany(L, 2);
      lua_pushboolean(L, lua_rawequal(L, 1, 2));
      return 1;
  }

  // rawlen(x) -> integer
  //
  // Returns the length of object x — which must be a table or a string —
  // without invoking the __len metamethod. Returns an integer.
  static int luaB_rawlen(lua_State* L) {
      int t = lua_type(L, 1);
      luaL_argexpected(L, t == LUA_TTABLE || t == LUA_TSTRING, 1,
                       "table or string");
      lua_pushinteger(L, l_castU2S(lua_rawlen(L, 1)));
      return 1;
  }

  // rawget(table, index) -> value
  //
  // Attempts to retrieve table[index], without using the __index metavalue.
  // The given index may be any value.
  static int luaB_rawget(lua_State* L) {
      luaL_checktype(L, 1, LUA_TTABLE);
      luaL_checkany(L, 2);
      lua_settop(L, 2);
      lua_rawget(L, 1);
      return 1;
  }

  // rawset(table, index, value) -> table
  //
  // Sets table[index] to value, without using the __newindex metavalue.
  // A given index may be any value different from nil and NaN, and
  // a given value may be any Lua value.
  //
  // Returns a table.
  static int luaB_rawset(lua_State* L) {
      luaL_checktype(L, 1, LUA_TTABLE);
      luaL_checkany(L, 2);
      luaL_checkany(L, 3);
      lua_settop(L, 3);
      lua_rawset(L, 1);
      return 1;
  }

  static int pushmode(lua_State* L, int oldmode) {
      if (oldmode == -1) {
          luaL_pushfail(L);
      } else {
          lua_pushstring(L, (oldmode == LUA_GCINC)
                            ? "incremental"
                            : "generational");
      }
      return 1;
  }

  // Check whether a call to lua_gc is valid — not inside a finalizer.
  #define checkvalres(res) \
      {                    \
          if (res == -1) { \
              break;       \
          }                \
      }

  // collectgarbage([option [, argument]])
  //
  // A generic interface to the garbage collector. Performs different functions
  // according to its first argument.
  static int luaB_collectgarbage(lua_State* L) {
      static const char* const opts[] = {
          "stop",      "restart",      "collect",     "count", "step",
          "isrunning", "generational", "incremental", "param", NULL
      };
      static const char optsnum[] = {
          LUA_GCSTOP,  LUA_GCRESTART, LUA_GCCOLLECT,
          LUA_GCCOUNT, LUA_GCSTEP,    LUA_GCISRUNNING,
          LUA_GCGEN,   LUA_GCINC,     LUA_GCPARAM
      };

      int o = optsnum[luaL_checkoption(L, 1, "collect", opts)];

      switch (o) {
          case LUA_GCCOUNT: {
              int k = lua_gc(L, o);
              int b = lua_gc(L, LUA_GCCOUNTB);
              checkvalres(k);
              lua_pushnumber(L, (lua_Number)k + ((lua_Number)b / 1024));
              return 1;
          }
          case LUA_GCSTEP: {
              lua_Integer n = luaL_optinteger(L, 2, 0);
              int res = lua_gc(L, o, cast_sizet(n));
              checkvalres(res);
              lua_pushboolean(L, res);
              return 1;
          }
          case LUA_GCISRUNNING: {
              int res = lua_gc(L, o);
              checkvalres(res);
              lua_pushboolean(L, res);
              return 1;
          }
          case LUA_GCGEN: {
              return pushmode(L, lua_gc(L, o));
          }
          case LUA_GCINC: {
              return pushmode(L, lua_gc(L, o));
          }
          case LUA_GCPARAM: {
              static const char* const params[] = {
                  "minormul", "majorminor", "minormajor", "pause",
                  "stepmul",  "stepsize",   NULL
              };
              static const char pnum[] = {
                  LUA_GCPMINORMUL,   LUA_GCPMAJORMINOR,
                  LUA_GCPMINORMAJOR, LUA_GCPPAUSE,
                  LUA_GCPSTEPMUL,    LUA_GCPSTEPSIZE
              };
              int p = pnum[luaL_checkoption(L, 2, NULL, params)];
              lua_Integer value = luaL_optinteger(L, 3, -1);
              lua_pushinteger(L, lua_gc(L, o, p, (int)value));
              return 1;
          }
          default: {
              int res = lua_gc(L, o);
              checkvalres(res);
              lua_pushinteger(L, res);
              return 1;
          }
      }
      // Invalid call. Inside finalizer
      luaL_pushfail(L);
      return 1;
  }

  // type(object) -> string
  //
  // Returns the type of its given object, encoded as a string.
  static int luaB_type(lua_State* L) {
      int t = lua_type(L, 1);
      luaL_argcheck(L, t != LUA_TNONE, 1, "value expected");
      lua_pushstring(L, lua_typename(L, t));
      return 1;
  }

  // next(table [, index]) -> index, value
  //
  // Allows a program to traverse all the fields of a given table.
  // Returns the next index of the table and its associated value.
  static int luaB_next(lua_State* L) {
      luaL_checktype(L, 1, LUA_TTABLE);
      // Create a second argument if there isn't one.
      lua_settop(L, 2);
      if (lua_next(L, 1)) {
          return 2;
      } else {
          lua_pushnil(L);
          return 1;
      }
  }

  static int pairscont(lua_State* L, int status, lua_KContext k) {
      (void)L;
      (void)status;
      (void)k;
      // __pairs did all the work. Return its results.
      return 4;
  }

  // pairs(table) -> __pairs(table) | next, table, nil, nil
  //
  // for key, value in pairs(table) do body end
  //
  // If the given table has a metamethod __pairs, calls __pairs(table) and returns
  // the first four results from the call. Otherwise, returns function "next",
  // the given table, plus two nil values.
  static int luaB_pairs(lua_State* L) {
      luaL_checkany(L, 1);
      // No metamethod?
      if (luaL_getmetafield(L, 1, "__pairs") == LUA_TNIL) {
          // generator
          lua_pushcfunction(L, luaB_next);
          // state
          lua_pushvalue(L, 1);
          // initial value
          lua_pushnil(L);
          // to-be-closed object
          lua_pushnil(L);
      } else {
          // Argument to "self" metamethod
          lua_pushvalue(L, 1);
          // Get four values from metamethod.
          lua_callk(L, 1, 4, 0, pairscont);
      }
      return 4;
  }

  // Traversal function for "ipairs".
  static int ipairsaux(lua_State* L) {
      lua_Integer i = luaL_checkinteger(L, 2);
      i = luaL_intop(+, i, 1);
      lua_pushinteger(L, i);
      return (lua_geti(L, 1, i) == LUA_TNIL) ? 1 : 2;
  }

  // ipairs(table) -> iterator, table, 0
  //
  // for index, value in ipairs(table) do body end
  static int luaB_ipairs(lua_State* L) {
      luaL_checkany(L, 1);
      // iteration function
      lua_pushcfunction(L, ipairsaux);
      // state
      lua_pushvalue(L, 1);
      // initial value
      lua_pushinteger(L, 0);
      return 3;
  }

  static int load_aux(lua_State* L, int status, int envidx) {
      if (l_likely(status == LUA_OK)) {
          // "env" parameter?
          if (envidx != 0) {
              // Environment for loaded function.
              lua_pushvalue(L, envidx);
              // Set it as the first upvalue.
              if (!lua_setupvalue(L, -2, 1)) {
                  // Remove "env" if not used by previous call.
                  lua_pop(L, 1);
              }
          }
          return 1;
      } else {
          luaL_pushfail(L);
          lua_insert(L, -2);
          return 2;
      }
  }

  static const char* getMode(lua_State* L, int idx) {
      const char* mode = luaL_optstring(L, idx, "bt");
      // Lua code cannot used fixed buffers.
      if (strchr(mode, 'B') != NULL) {
          luaL_argerror(L, idx, "invalid mode");
      }
      return mode;
  }

  // loadfile ([filename [, mode [, environment]]])
  //
  // Compiles a chunk from the given file. Defaults to the standard input
  // if no file name is given.
  static int luaB_loadfile(lua_State* L) {
      const char* fname = luaL_optstring(L, 1, NULL);
      const char* mode = getMode(L, 2);
      int env = (!lua_isnone(L, 3) ? 3 : 0);
      int status = luaL_loadfilex(L, fname, mode);
      return load_aux(L, status, env);
  }

  // === Generic Read Function ===

  // Reserved slot protects buffered string from garbage collection during parsing.
  #define RESERVEDSLOT 5

  static const char* generic_reader(lua_State* L, void* ud, size_t* size) {
      (void)(ud);
      luaL_checkstack(L, 2, "too many nested functions");
      // Get function.
      lua_pushvalue(L, 1);
      // Call it.
      lua_call(L, 0, 1);
      if (lua_isnil(L, -1)) {
          // Pop the result.
          lua_pop(L, 1);
          *size = 0;
          return NULL;
      } else if (l_unlikely(!lua_isstring(L, -1))) {
          luaL_error(L, "reader function must return a string");
      }
      // Save string in reserved slot.
      lua_replace(L, RESERVEDSLOT);
      return lua_tolstring(L, RESERVEDSLOT, size);
  }

  // load(chunk [, name [, mode [, environment]]]) -> function, nil | nil, error
  //
  // Loads a chunk. A chunk is either a string or a reader function, which provides
  // a string piecemeal.
  //
  // If successful returns a function. Otherwise, returns nil plus an error message.
  static int luaB_load(lua_State* L) {
      int status;
      size_t l;
      const char* s = lua_tolstring(L, 1, &l);
      const char* mode = getMode(L, 3);
      // "env" index or 0 if no "env".
      int env = (!lua_isnone(L, 4) ? 4 : 0);
      // Loading a string?
      if (s != NULL) {
          const char* chunkname = luaL_optstring(L, 2, s);
          status = luaL_loadbufferx(L, s, l, chunkname, mode);
      }
      // Loading from a reader function.
      else {
          const char* chunkname = luaL_optstring(L, 2, "=(load)");
          luaL_checktype(L, 1, LUA_TFUNCTION);
          lua_settop(L, RESERVEDSLOT);
          status = lua_load(L, generic_reader, NULL, chunkname, mode);
      }
      return load_aux(L, status, env);
  }

  static int dofilecont(lua_State* L, int d1, lua_KContext d2) {
      (void)d1;
      (void)d2; /* only to match 'lua_Kfunction' prototype */
      return lua_gettop(L) - 1;
  }

  // dofile([name]) -> ...
  //
  // Opens the given file and executes its contents as a Lua chunk. Returns all
  // values returned by the chunk. When called without arguments, dofile executes
  // the content of the standard input. In case of errors, propagates the
  // error to its caller.
  static int luaB_dofile(lua_State* L) {
      const char* fname = luaL_optstring(L, 1, NULL);
      lua_settop(L, 1);
      if (l_unlikely(luaL_loadfile(L, fname) != LUA_OK)) {
          return lua_error(L);
      }
      lua_callk(L, 0, LUA_MULTRET, 0, dofilecont);
      return dofilecont(L, 0, 0);
  }

  // assert(expression [, message]) -> ... | error
  //
  // Raises an error if the value of its given expression is falsey.
  // Otherwise, returns all the results of the given expression.
  //
  // truthy -> ...
  // falsey -> error
  static int luaB_assert(lua_State* L) {
      // Condition true? Return all arguments.
      if (l_likely(lua_toboolean(L, 1))) {
          return lua_gettop(L);
      }
      // Error.
      else {
          // Remove condition.
          luaL_checkany(L, 1);
          lua_remove(L, 1);
          // Default message.
          lua_pushliteral(L, "assertion failed!");
          // Leave only message. Default if no other.
          lua_settop(L, 1);
          // Call error.
          return luaB_error(L);
      }
  }

  // select(index, ...) -> ... | number
  // y, z  = select(2, "x", "y", "z")
  // three = select("#", "x", "y", "z")
  //
  // If index is a number, returns all arguments after the given index.
  // A negative number is offset from the end. If index is the string "#",
  // returns the total number of extra arguments given.
  static int luaB_select(lua_State* L) {
      int n = lua_gettop(L);
      if (lua_type(L, 1) == LUA_TSTRING && *lua_tostring(L, 1) == '#') {
          lua_pushinteger(L, n - 1);
          return 1;
      } else {
          lua_Integer i = luaL_checkinteger(L, 1);
          if (i < 0) {
              i = n + i;
          } else if (i > n) {
              i = n;
          }
          luaL_argcheck(L, 1 <= i, 1, "index out of range");
          return n - (int)i;
      }
  }

  // Continuation function for "pcall" and "xpcall".
  static int finishpcall(lua_State* L, int status, lua_KContext extra) {
      // Error?
      if (l_unlikely(status != LUA_OK && status != LUA_YIELD)) {
          // First result false.
          lua_pushboolean(L, 0);
          // Error message.
          lua_pushvalue(L, -2);
          // Return false, message
          return 2;
      } else {
          // Return all results.
          return lua_gettop(L) - (int)extra;
      }
  }

  // pcall(function [, argument, ...]) -> true, result, ... | false, error
  //
  // Calls the function with the given arguments in protected mode.
  // Catches errors inside the given function and returns a status code
  // alongside either its results or a captured error object.
  static int luaB_pcall(lua_State* L) {
      int status;
      luaL_checkany(L, 1);
      // First result if no errors.
      lua_pushboolean(L, 1);
      lua_insert(L, 1);
      status = lua_pcallk(L, lua_gettop(L) - 2, LUA_MULTRET, 0, 0, finishpcall);
      return finishpcall(L, status, 0);
  }

  // xpcall (function, message-handler [, argument, ...])
  //
  // This function is similar to "pcall", except that it sets a new message handler.
  static int luaB_xpcall(lua_State* L) {
      int status;
      int n = lua_gettop(L);
      // Check error function.
      luaL_checktype(L, 2, LUA_TFUNCTION);
      // First result.
      lua_pushboolean(L, 1);
      // Function.
      lua_pushvalue(L, 1);
      // Move them below function's arguments.
      lua_rotate(L, 3, 2);
      status = lua_pcallk(L, n - 2, LUA_MULTRET, 2, 2, finishpcall);
      return finishpcall(L, status, 2);
  }

  // tostring(value) -> string
  //
  // Converts a value of any type to a string in a human-readable format.
  //
  // If the metatable of the given value has a "__tostring field", calls the
  // corresponding metatable with the given value as argument. May use "__name"
  // field if the associated value is a string.
  static int luaB_tostring(lua_State* L) {
      luaL_checkany(L, 1);
      luaL_tolstring(L, 1, NULL);
      return 1;
  }

  // Populate the global table with Lua's base library.
  LUAMOD_API int luaopen_base(lua_State* L) {
      lua_pushglobaltable(L);
      luaL_setfuncs(L, base_funcs, 0);
      lua_pushvalue(L, -1);
      lua_setfield(L, -2, LUA_GNAME);
      lua_pushliteral(L, LUA_VERSION);
      lua_setfield(L, -2, "_VERSION");
      return 1;
  }
